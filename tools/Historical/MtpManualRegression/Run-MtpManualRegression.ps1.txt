function Add-ReportLine([string]$line = '') {
    Add-Content -LiteralPath $reportPath -Value $line -Encoding UTF8
}

function Ask-Result([string]$id, [string]$question) {
    while ($true) {
        Write-Host ''
        Write-Host $question -ForegroundColor Cyan
        Write-Host '  [P] 通过   [F] 失败   [U] 未测/无法判断' -ForegroundColor DarkGray
        $answer = (Read-Host '输入 P、F 或 U').Trim().ToUpperInvariant()
        if ($answer -in @('P','F','U')) {
            $results[$id] = switch ($answer) { 'P' { '通过' } 'F' { '失败' } default { '未测' } }
            $notes[$id] = Read-Host '补充说明（可直接回车）'
            return
        }
        Write-Host '请输入 P、F 或 U。' -ForegroundColor Yellow
    }
}

function Invoke-LoggedCommand([string]$name, [string]$filePath, [string[]]$arguments) {
    $log = Join-Path $evidenceRoot "$name.log"
    Write-Host "正在执行 $name ..." -ForegroundColor Yellow
    & $filePath @arguments *> $log
    $exitCode = $LASTEXITCODE
    $script:results["自动化_$name"] = if ($exitCode -eq 0) { '通过' } else { "失败（退出码 $exitCode）" }
    Add-ReportLine "- $name：$($script:results["自动化_$name"])；日志：$log"
}
